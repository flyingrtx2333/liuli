cmake_minimum_required(VERSION 3.20)
project(ExplicitUI LANGUAGES CXX)

find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
option(EUI_LINUX_ACCESSIBILITY "Experimental Linux ATK bridge (opt-in at runtime)" OFF)
set(EUI_A11Y_SDK "" CACHE PATH "Extracted Debian SDK root, or empty for system pkg-config")
option(EUI_BUILD_WINDOW "Build the SDL2/Cairo native window" ON)
option(EUI_LINUX_RUNTIME_FALLBACK "Allow Linux shared runtimes without development headers" ON)
set(EUI_MODEL "${CMAKE_CURRENT_SOURCE_DIR}/examples/counter_list.eui.json" CACHE FILEPATH "Model AST")
set(EUI_LAYOUT "${CMAKE_CURRENT_SOURCE_DIR}/examples/responsive.layout.json" CACHE FILEPATH "Layout AST")
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${generated}")
add_custom_command(OUTPUT "${generated}/scripted.cpp" "${generated}/model.h"
  COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/compiler.py" "${EUI_MODEL}"
          -o "${generated}/scripted.cpp" --model-header "${generated}/model.h"
  DEPENDS compiler.py "${EUI_MODEL}" VERBATIM)
add_custom_target(eui_model DEPENDS "${generated}/scripted.cpp" "${generated}/model.h")

function(eui_options target)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE "${generated}" "${CMAKE_CURRENT_SOURCE_DIR}/native")
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()

add_executable(eui-scripted "${generated}/scripted.cpp")
add_dependencies(eui-scripted eui_model)
eui_options(eui-scripted)
add_executable(eui-live native/live_host.cpp)
add_dependencies(eui-live eui_model)
eui_options(eui-live)

if(EUI_BUILD_WINDOW)
  find_package(Threads REQUIRED)
  find_package(PkgConfig QUIET)
  find_package(SDL2 CONFIG QUIET)
  if(TARGET SDL2::SDL2)
    set(eui_sdl SDL2::SDL2)
  elseif(PkgConfig_FOUND)
    pkg_check_modules(SDL2_PC QUIET IMPORTED_TARGET sdl2)
    if(TARGET PkgConfig::SDL2_PC)
      set(eui_sdl PkgConfig::SDL2_PC)
    endif()
  endif()
  if(PkgConfig_FOUND)
    pkg_check_modules(PANGO_PC QUIET IMPORTED_TARGET pangocairo)
    if(TARGET PkgConfig::PANGO_PC)
      set(eui_pango PkgConfig::PANGO_PC)
    endif()
    pkg_check_modules(CAIRO_PC QUIET IMPORTED_TARGET cairo)
    if(TARGET PkgConfig::CAIRO_PC)
      set(eui_cairo PkgConfig::CAIRO_PC)
    endif()
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND EUI_LINUX_RUNTIME_FALLBACK)
    if(NOT eui_sdl)
      find_library(EUI_SDL_RUNTIME NAMES SDL2-2.0 libSDL2-2.0.so.0 REQUIRED)
      set(eui_sdl "${EUI_SDL_RUNTIME}")
      message(STATUS "Using Linux SDL2 runtime ABI fallback: ${eui_sdl}")
    endif()
    if(NOT eui_pango)
      find_library(EUI_PANGOCAIRO_RUNTIME NAMES pangocairo-1.0 libpangocairo-1.0.so.0 REQUIRED)
      find_library(EUI_PANGO_RUNTIME NAMES pango-1.0 libpango-1.0.so.0 REQUIRED)
      find_library(EUI_GOBJECT_RUNTIME NAMES gobject-2.0 libgobject-2.0.so.0 REQUIRED)
      find_library(EUI_GLIB_RUNTIME NAMES glib-2.0 libglib-2.0.so.0 REQUIRED)
      set(eui_pango "${EUI_PANGOCAIRO_RUNTIME}" "${EUI_PANGO_RUNTIME}" "${EUI_GOBJECT_RUNTIME}" "${EUI_GLIB_RUNTIME}")
      message(STATUS "Using Linux Pango runtime ABI fallback")
    endif()
    if(NOT eui_cairo)
      find_library(EUI_CAIRO_RUNTIME NAMES cairo libcairo.so.2 REQUIRED)
      set(eui_cairo "${EUI_CAIRO_RUNTIME}")
      message(STATUS "Using Linux Cairo runtime ABI fallback: ${eui_cairo}")
    endif()
  endif()
  if(NOT eui_sdl OR NOT eui_cairo OR NOT eui_pango)
    message(FATAL_ERROR "Native window needs SDL2 development config/pkg-config and Cairo/PangoCairo pkg-config. Configure with -DEUI_BUILD_WINDOW=OFF for the core only.")
  endif()
  add_custom_command(OUTPUT "${generated}/scene.h"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/layout_compiler.py" "${EUI_LAYOUT}"
            --model "${EUI_MODEL}" -o "${generated}/scene.h"
    DEPENDS layout_compiler.py compiler.py "${EUI_LAYOUT}" "${EUI_MODEL}" VERBATIM)
  add_executable(eui-window native/window_host.cpp native/glass_renderer.cpp native/shaped_text.cpp "${generated}/scene.h")
  add_dependencies(eui-window eui_model)
  eui_options(eui-window)
  target_link_libraries(eui-window PRIVATE ${eui_sdl} ${eui_cairo} ${eui_pango} Threads::Threads)
  if(EUI_LINUX_ACCESSIBILITY)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
      message(FATAL_ERROR "ATK accessibility spike requires Linux")
    endif()
    add_library(eui-accessibility STATIC native/linux_accessibility.cpp)
    eui_options(eui-accessibility)
    if(EUI_A11Y_SDK)
      target_include_directories(eui-accessibility SYSTEM PRIVATE "${EUI_A11Y_SDK}/usr/include/atk-1.0" "${EUI_A11Y_SDK}/usr/include/at-spi2-atk/2.0" "${EUI_A11Y_SDK}/usr/include/glib-2.0" "${EUI_A11Y_SDK}/usr/lib/x86_64-linux-gnu/glib-2.0/include")
      find_library(EUI_ATK NAMES libatk-1.0.so.0 REQUIRED)
      find_library(EUI_ATK_BRIDGE NAMES libatk-bridge-2.0.so.0 REQUIRED)
      find_library(EUI_GOBJECT NAMES libgobject-2.0.so.0 REQUIRED)
      find_library(EUI_GLIB NAMES libglib-2.0.so.0 REQUIRED)
      target_link_libraries(eui-accessibility PUBLIC ${EUI_ATK} ${EUI_ATK_BRIDGE} ${EUI_GOBJECT} ${EUI_GLIB} Threads::Threads)
    else()
      if(NOT PkgConfig_FOUND)
        message(FATAL_ERROR "Accessibility needs pkg-config or EUI_A11Y_SDK")
      endif()
      pkg_check_modules(ATK REQUIRED IMPORTED_TARGET "atk>=2.62.1" "atk-bridge-2.0>=2.62.1")
      target_link_libraries(eui-accessibility PUBLIC PkgConfig::ATK Threads::Threads)
    endif()
    target_link_libraries(eui-accessibility PUBLIC ${CMAKE_DL_LIBS})
    add_executable(eui-accessibility-runtime-check tests/native/accessibility_runtime_check.cpp)
    eui_options(eui-accessibility-runtime-check)
    target_link_libraries(eui-accessibility-runtime-check PRIVATE eui-accessibility)
    add_executable(eui-atspi-probe tests/native/atspi_probe.cpp)
    eui_options(eui-atspi-probe)
    if(EUI_A11Y_SDK)
      target_include_directories(eui-atspi-probe SYSTEM PRIVATE "${EUI_A11Y_SDK}/usr/include/at-spi-2.0" "${EUI_A11Y_SDK}/usr/include/dbus-1.0" "${EUI_A11Y_SDK}/usr/lib/x86_64-linux-gnu/dbus-1.0/include" "${EUI_A11Y_SDK}/usr/include/glib-2.0" "${EUI_A11Y_SDK}/usr/lib/x86_64-linux-gnu/glib-2.0/include")
      find_library(EUI_ATSPI NAMES libatspi.so.0 REQUIRED)
      target_link_libraries(eui-atspi-probe PRIVATE ${EUI_ATSPI} ${EUI_GOBJECT} ${EUI_GLIB})
    else()
      pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET "atspi-2>=2.62.1")
      target_link_libraries(eui-atspi-probe PRIVATE PkgConfig::ATSPI)
    endif()
    add_executable(eui-accessibility-session-test tests/native/accessibility_session_test.cpp)
    eui_options(eui-accessibility-session-test)
    target_link_libraries(eui-accessibility-session-test PRIVATE eui-accessibility)
    add_executable(eui-accessibility-tree-test tests/native/accessibility_tree_test.cpp)
    eui_options(eui-accessibility-tree-test)
    target_include_directories(eui-accessibility-tree-test SYSTEM PRIVATE $<TARGET_PROPERTY:eui-accessibility,INCLUDE_DIRECTORIES>)
    target_link_libraries(eui-accessibility-tree-test PRIVATE eui-accessibility)
    target_compile_definitions(eui-window PRIVATE EUI_LINUX_ACCESSIBILITY=1)
    target_link_libraries(eui-window PRIVATE eui-accessibility)
  endif()
  if(TARGET SDL2::SDL2main)
    target_link_libraries(eui-window PRIVATE SDL2::SDL2main)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_test(NAME scripted_smoke COMMAND eui-scripted)
  set_tests_properties(scripted_smoke PROPERTIES TIMEOUT 10)
  if(EUI_LINUX_ACCESSIBILITY AND EUI_BUILD_WINDOW)
    add_test(NAME accessibility_tree COMMAND eui-accessibility-tree-test)
  endif()
  if(EUI_BUILD_WINDOW)
    get_filename_component(eui_test_model "${EUI_MODEL}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    get_filename_component(eui_test_layout "${EUI_LAYOUT}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
    if(eui_test_model STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/examples/counter_list.eui.json"
       AND eui_test_layout STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/examples/responsive.layout.json")
      add_test(NAME window_synthetic COMMAND eui-window --smoke "${CMAKE_CURRENT_BINARY_DIR}/synthetic.png")
      set_tests_properties(window_synthetic PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy" TIMEOUT 30)
    else()
      message(STATUS "Skipping bundled counter-window smoke for custom declarations; provide application-specific interaction tests.")
    endif()
  endif()
endif()
